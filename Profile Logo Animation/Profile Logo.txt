/**
 * ProfileLogo — animated official PROFILE logo
 *
 * @param {string}  state  - "loading" | "connecting" | "syncing" | "success" |
 *                           "offline" | "error" | "retry" | "searching" |
 *                           "processing" | "idle"
 * @param {number}  size   - rendered width in px (height auto-calculated, default 40)
 * @param {string}  label  - optional aria-label override
 */
function ProfileLogo({ state = "loading", size = 40, label }) {
  // viewBox constants — must match the HTML file exactly
  const VW = 400, VH = 420;
  const height = Math.round(size * VH / VW);

  // Head geometry — must match the HTML file exactly
  const HX = 202.36, HY = 150.21, HR = 37.98;
  const R1 = 55.0,   R2 = 80.0,   R3 = 108.0;

  // Arc path helper — right-facing partial circle (the Wi-Fi bars)
  function arcPath(r, deg = 65) {
    const a = (deg * Math.PI) / 180;
    const x1 = HX + r * Math.cos(-a), y1 = HY + r * Math.sin(-a);
    const x2 = HX + r * Math.cos(a),  y2 = HY + r * Math.sin(a);
    return `M${x1.toFixed(2)},${y1.toFixed(2)} A${r},${r} 0 0 1 ${x2.toFixed(2)},${y2.toFixed(2)}`;
  }

  const A1 = arcPath(R1), A2 = arcPath(R2), A3 = arcPath(R3);
  const TO = (x, y) => ({ transformOrigin: `${x}px ${y}px` });

  // ── ANIMATION LAYERS PER STATE ──────────────────────────────
  // behind: rendered before <use href="#logo-img"/>  (glow, rings)
  // over:   rendered after  <use href="#logo-img"/>  (arc bars, icons)

  let behind = null;
  let over   = null;
  let groupStyle = {};

  if (state === "loading") {
    behind = <circle cx={HX} cy={HY} r={HR+16} fill="url(#gg)" style={{animation:"glow-breathe 2s ease-in-out infinite"}}/>;
    over   = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 2s cubic-bezier(.4,0,.2,1) infinite",...TO(HX,HY)}}/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 2s cubic-bezier(.4,0,.2,1) .4s infinite",...TO(HX,HY)}}/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 2s cubic-bezier(.4,0,.2,1) .8s infinite",...TO(HX,HY)}}/>
    </>);
  }

  else if (state === "connecting") {
    behind = <circle cx={HX} cy={HY} r={HR+16} fill="url(#gg)" style={{animation:"glow-breathe 1.5s ease-in-out infinite"}}/>;
    over   = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.5s cubic-bezier(.4,0,.2,1) infinite",...TO(HX,HY)}}/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.5s cubic-bezier(.4,0,.2,1) .28s infinite",...TO(HX,HY)}}/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.5s cubic-bezier(.4,0,.2,1) .56s infinite",...TO(HX,HY)}}/>
    </>);
  }

  else if (state === "syncing") {
    const oc = 2 * Math.PI * (HR + 20);
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+20} fill="none" stroke="#00E5FF" strokeWidth="1.5" strokeDasharray={`${(oc*.22).toFixed(1)} ${(oc*.78).toFixed(1)}`} strokeLinecap="round" opacity="0" style={{animation:"orbit-spin 1.4s linear infinite",...TO(HX,HY)}}/>
      <circle cx={HX} cy={HY} r={HR+16} fill="url(#gg)" style={{animation:"glow-breathe 1.2s ease-in-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.2s cubic-bezier(.4,0,.2,1) infinite",...TO(HX,HY)}}/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.2s cubic-bezier(.4,0,.2,1) .22s infinite",...TO(HX,HY)}}/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.2s cubic-bezier(.4,0,.2,1) .44s infinite",...TO(HX,HY)}}/>
    </>);
  }

  else if (state === "success") {
    const cx2 = HX+20, cy2 = HY+20;
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+16} fill="url(#gg)" style={{animation:"glow-breathe 2s ease-in-out infinite"}}/>
      <circle cx={HX} cy={HY} r={HR+8}  fill="none" stroke="#00C853" strokeWidth="2.5" opacity="0" style={{animation:"success-ring 2.5s ease-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round"/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round"/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round"/>
      <circle cx={cx2} cy={cy2} r="12" fill="#00C853"/>
      <path d={`M${cx2-7},${cy2} L${cx2-1},${cy2+6} L${cx2+8},${cy2-6}`} fill="none" stroke="white" strokeWidth="2.3" strokeLinecap="round" strokeLinejoin="round" strokeDasharray="24" style={{animation:"check-draw 2.5s ease infinite"}}/>
    </>);
  }

  else if (state === "offline") {
    const cx2 = HX+20, cy2 = HY+20;
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+18} fill="none" stroke="#374151" strokeWidth="1.5" strokeDasharray="6 5" opacity="0.55"/>
      <circle cx={HX} cy={HY} r={HR+14} fill="url(#gg)" opacity="0.18" style={{animation:"glow-dim 3s ease-in-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="#4B5563" strokeWidth="5.5" strokeLinecap="round" style={{animation:"glow-dim 2.5s ease-in-out infinite"}}/>
      <path d={A2} fill="none" stroke="#374151" strokeWidth="4.5" strokeLinecap="round" style={{animation:"glow-dim 2.5s ease-in-out .4s infinite"}}/>
      <path d={A3} fill="none" stroke="#1F2937" strokeWidth="3.5" strokeLinecap="round" style={{animation:"glow-dim 2.5s ease-in-out .8s infinite"}}/>
      <circle cx={cx2} cy={cy2} r="12" fill="none" stroke="#F44336" strokeWidth="2.2"/>
      <line x1={cx2-7} y1={cy2-7} x2={cx2+7} y2={cy2+7} stroke="#F44336" strokeWidth="2.2" strokeLinecap="round"/>
      <line x1={cx2+7} y1={cy2-7} x2={cx2-7} y2={cy2+7} stroke="#F44336" strokeWidth="2.2" strokeLinecap="round"/>
    </>);
  }

  else if (state === "error") {
    const tx = HX+20, ty = HY+7;
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+22} fill="url(#eg)" opacity="0" style={{animation:"error-bg 2s ease-in-out infinite"}}/>
      <circle cx={HX} cy={HY} r={HR+9}  fill="none" stroke="#F44336" strokeWidth="2.5" opacity="0" style={{animation:"error-ring 2s ease-in-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="#EF4444" strokeWidth="5.5" strokeLinecap="round" opacity="0.65"/>
      <path d={A2} fill="none" stroke="#EF4444" strokeWidth="4.5" strokeLinecap="round" opacity="0.28"/>
      <polygon points={`${tx},${ty} ${tx+12},${ty+20} ${tx-12},${ty+20}`} fill="#FF9800" opacity="0" style={{animation:"amber-flash 2s ease-in-out .15s infinite"}}/>
      <line x1={tx} y1={ty+5} x2={tx} y2={ty+12} stroke="white" strokeWidth="2.2" strokeLinecap="round" opacity="0" style={{animation:"amber-flash 2s ease-in-out .15s infinite"}}/>
      <circle cx={tx} cy={ty+16} r="1.5" fill="white" opacity="0" style={{animation:"amber-flash 2s ease-in-out .15s infinite"}}/>
    </>);
    groupStyle = { animation: "shake-logo .65s cubic-bezier(.36,.07,.19,.97) .3s infinite" };
  }

  else if (state === "retry") {
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+18} fill="none" stroke="#1E3A8A" strokeWidth="1.2" strokeDasharray="4 5" opacity="0.45" style={{animation:"orbit-spin 3s linear infinite",...TO(HX,HY)}}/>
      <circle cx={HX} cy={HY} r={HR+14} fill="url(#gg)" style={{animation:"glow-breathe 1.8s ease-in-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0" style={{animation:"arc-grow 1.8s cubic-bezier(.4,0,.6,1) infinite",...TO(HX,HY)}}/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0" style={{animation:"arc-grow 1.8s cubic-bezier(.4,0,.6,1) .3s infinite",...TO(HX,HY)}}/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0" style={{animation:"arc-grow 1.8s cubic-bezier(.4,0,.6,1) .6s infinite",...TO(HX,HY)}}/>
    </>);
  }

  else if (state === "searching") {
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+28} fill="none" stroke="#1E3A8A" strokeWidth="0.8" opacity="0.35"/>
      <line x1={HX} y1={HY} x2={HX} y2={HY-HR-28} stroke="#00E5FF" strokeWidth="1.8" opacity="0.5" style={{animation:"scan-rotate 2.5s linear infinite",...TO(HX,HY)}}/>
      <circle cx={HX} cy={HY} r={HR+14} fill="url(#gg)" style={{animation:"glow-dim 2.5s ease-in-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 2.4s ease-in-out infinite",...TO(HX,HY)}}/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 2.4s ease-in-out .6s infinite",...TO(HX,HY)}}/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 2.4s ease-in-out 1.2s infinite",...TO(HX,HY)}}/>
    </>);
  }

  else if (state === "processing") {
    const dc = 2 * Math.PI * (HR + 19);
    behind = (<>
      <circle cx={HX} cy={HY} r={HR+19} fill="none" stroke="#00E5FF" strokeWidth="2.8" strokeDasharray={`${(dc*.30).toFixed(1)} ${(dc*.70).toFixed(1)}`} strokeLinecap="round" opacity="0.8" style={{animation:"process-spin 1s linear infinite",...TO(HX,HY)}}/>
      <circle cx={HX} cy={HY} r={HR+14} fill="url(#gg)" style={{animation:"glow-breathe 1.3s ease-in-out infinite"}}/>
    </>);
    over = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.3s ease-in-out infinite",...TO(HX,HY)}}/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.3s ease-in-out .22s infinite",...TO(HX,HY)}}/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0" style={{animation:"arc-wave 1.3s ease-in-out .44s infinite",...TO(HX,HY)}}/>
    </>);
  }

  else { // idle (default)
    behind = <circle cx={HX} cy={HY} r={HR+14} fill="url(#gg)" opacity="0.3" style={{animation:"glow-idle 4s ease-in-out infinite"}}/>;
    over   = (<>
      <path d={A1} fill="none" stroke="url(#ag)" strokeWidth="5.5" strokeLinecap="round" opacity="0.22"/>
      <path d={A2} fill="none" stroke="url(#ag)" strokeWidth="4.5" strokeLinecap="round" opacity="0.10"/>
      <path d={A3} fill="none" stroke="url(#ag)" strokeWidth="3.5" strokeLinecap="round" opacity="0.04"/>
    </>);
  }

  return (
    <svg
      width={size}
      height={height}
      viewBox={`0 0 ${VW} ${VH}`}
      xmlns="http://www.w3.org/2000/svg"
      xmlnsXlink="http://www.w3.org/1999/xlink"
      role="img"
      aria-label={label || `PROFILE logo – ${state}`}
      style={{ display: "block", flexShrink: 0 }}
    >
      <g style={groupStyle}>
        {behind}
        <use href="#logo-img" />
        {over}
      </g>
    </svg>
  );
}
